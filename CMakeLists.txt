cmake_minimum_required(VERSION 3.20)
project(ice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 11)
endif()
add_subdirectory(Platform)
add_subdirectory(Engine)
add_subdirectory(Projects)
include(CTest)
if(BUILD_TESTING AND NOT WIN32)
    add_executable(ice_core_runtime_tests tests/core_runtime.cpp)
    target_link_libraries(ice_core_runtime_tests PRIVATE ICEngine)
    add_test(NAME ice_core_runtime COMMAND ice_core_runtime_tests)
endif()
if(BUILD_TESTING)
    file(GLOB_RECURSE AIRHOCKEY_PHYSICS CONFIGURE_DEPENDS Projects/Airhockey/Box2D/*.cpp)
    add_executable(airhockey_gameplay_tests tests/airhockey_gameplay.cpp Projects/Airhockey/Source/AiInput.cpp ${AIRHOCKEY_PHYSICS})
    target_include_directories(airhockey_gameplay_tests PRIVATE Projects/Airhockey/Source Projects/Airhockey)
    target_link_libraries(airhockey_gameplay_tests PRIVATE ICEngine)
    add_test(NAME airhockey_gameplay COMMAND airhockey_gameplay_tests)
endif()
if(APPLE AND BUILD_TESTING)
    add_executable(ice_platform_smoke tests/platform_osx.mm)
    target_link_libraries(ice_platform_smoke PRIVATE ICEngine)
    set_target_properties(ice_platform_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    # This test requires a real WindowServer session; run it separately from headless CTest.
endif()
if(UNIX AND NOT APPLE AND BUILD_TESTING)
    add_executable(ice_platform_smoke tests/platform_x11.cpp)
    target_link_libraries(ice_platform_smoke PRIVATE ICEngine)
    set_target_properties(ice_platform_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(WIN32 AND BUILD_TESTING)
    add_executable(ice_platform_smoke tests/platform_win.cpp)
    target_link_libraries(ice_platform_smoke PRIVATE ICEngine)
    set_target_properties(ice_platform_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(BUILD_TESTING)
    add_executable(ice_playground_tests tests/playground_gameplay.cpp)
    target_include_directories(ice_playground_tests PRIVATE Projects/Playgrounds/Source)
    add_test(NAME ice_playground_gameplay COMMAND ice_playground_tests)
endif()

if(BUILD_TESTING)
    add_executable(ice_host_tests tests/host_runtime.cpp)
    target_link_libraries(ice_host_tests PRIVATE ICEPlatformRuntime)
    add_test(NAME ice_host_runtime COMMAND ice_host_tests)
endif()
