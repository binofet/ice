cmake_minimum_required(VERSION 3.22)
project(ice LANGUAGES CXX)
include(cmake/ICEStandards.cmake)
include(CTest)
set(desktop_default ON)
if(ANDROID OR CMAKE_SYSTEM_NAME MATCHES "^(iOS|watchOS|tvOS|visionOS)$")
    set(desktop_default OFF)
endif()
option(ICE_BUILD_DESKTOP "Build native AppKit/Win32/X11 windows without a graphics context" ${desktop_default})
option(ICE_BUILD_TOOLS "Build headless companion replay/host tools" ON)
set(metal_default OFF)
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "watchOS")
    set(metal_default ON)
endif()
option(ICE_BUILD_METAL "Build the native Metal presentation module" ${metal_default})
option(ICE_BUILD_OPENGL "Build the native OpenGL 4.1+ core presentation module" ${ICE_BUILD_DESKTOP})

    enable_language(OBJCXX)
endif()

if(BUILD_TESTING)

        add_executable(${target} ${source})
        target_link_libraries(${target} PRIVATE ${module})
        ice_strict_target(${target})
    endfunction()
    ice_test(ice_host_tests tests/host_runtime.cpp ice::platform)
    add_test(NAME ice_host_runtime COMMAND ice_host_tests)
    ice_test(ice_math_tests tests/math.cpp ice::math)

    ice_test(ice_sha256_tests tests/sha256.cpp ice::expected)
    target_include_directories(ice_sha256_tests PRIVATE engine/include)
    add_test(NAME ice_sha256 COMMAND ice_sha256_tests)
    ice_test(ice_primitive_mesh_tests tests/primitive_mesh.cpp ice::assets)
    add_test(NAME ice_primitive_mesh COMMAND ice_primitive_mesh_tests)
    ice_test(ice_gltf_asset_tests tests/gltf_asset.cpp ice::assets)
    add_test(NAME ice_gltf_assets COMMAND ice_gltf_asset_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/gltf" "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/gltf/DamagedHelmet/DamagedHelmet.glb")
    ice_test(ice_resource_uri_tests tests/resource_uri.cpp ice::assets)
    add_test(NAME ice_resource_uri COMMAND ice_resource_uri_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/gltf/Interleaved")

    add_test(NAME ice_scene_animation COMMAND ice_scene_animation_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/gltf/SkinProbe" "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/gltf/Fox/Fox.glb")

    ice_test(ice_data_graph_tests tests/data_graph.cpp ice::data)

    add_test(NAME ice_data_graph COMMAND ice_data_graph_tests "${CMAKE_CURRENT_SOURCE_DIR}/Tools/MetagenIntegration/Generated")
    ice_test(ice_data_json_tests tests/data_json.cpp ice::data)
    add_test(NAME ice_data_json COMMAND ice_data_json_tests)

    endif()

    add_library(ice_public_headers OBJECT tests/public_headers.cpp)
    target_link_libraries(ice_public_headers PRIVATE ice::engine)
    ice_strict_target(ice_public_headers)
    if(TARGET ice::shader_sources)
        ice_test(ice_shader_snapshot_tests tests/shader_snapshot.cpp ice::shader_sources)
        add_test(NAME ice_shader_snapshot COMMAND ice_shader_snapshot_tests)

    if(TARGET ice::render)
        ice_test(ice_mesh_tests tests/mesh.cpp ice::render)

        target_compile_definitions(ice_mesh_tests PRIVATE
