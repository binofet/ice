project(Engine)


file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "icWindow(OSX|WIN|X11)\\.cpp$|icInput(OSX|WIN|X11)\\.cpp$|icFileWin\\.cpp$|icFontDX9\\.cpp$|icGXDevice(DX9|DX11|GLES|GLES2)\\.cpp$|ic(VertBuffer|IndexBuf|Texture)(DX9|GLES)\\.cpp$|Core/Events/|ObjectModel/|Scripting/")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS 3rdparty/Lua/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
if(APPLE)
  list(APPEND ENGINE_SOURCES src/Core/Application/Window/icWindowOSX.mm src/Core/IO/icInputOSX.cpp)
elseif(WIN32)
  list(APPEND ENGINE_SOURCES src/Core/Application/Window/icWindowWIN.cpp src/Core/IO/icInputWIN.cpp src/Core/IO/icFileWin.cpp)
else()
  list(APPEND ENGINE_SOURCES src/Core/Application/Window/icWindowX11.cpp src/Core/IO/icInputX11.cpp)
endif()
add_library(ICEngine ${ENGINE_SOURCES} ${LUA_SOURCES} 3rdparty/TLSF/tlsf.c 3rdparty/lea_malloc/leamalloc.c)
target_include_directories(ICEngine PUBLIC include PRIVATE src 3rdparty)
if(APPLE)
  target_compile_definitions(ICEngine PUBLIC GL_SILENCE_DEPRECATION)
  target_link_libraries(ICEngine PUBLIC "-framework AppKit" "-framework ApplicationServices" "-framework OpenGL" "-framework OpenAL" "-framework CoreText" "-framework CoreGraphics")
endif()

if(NOT APPLE)
  find_package(OpenGL REQUIRED)
  find_package(OpenAL REQUIRED)
  target_include_directories(ICEngine PUBLIC ${OPENAL_INCLUDE_DIR})
  target_link_libraries(ICEngine PUBLIC OpenGL::GL OpenGL::GLU ${OPENAL_LIBRARY})
  if(WIN32)
    target_compile_definitions(ICEngine PUBLIC ICE_OPENGL_ONLY WIN32 _CRT_SECURE_NO_WARNINGS NOMINMAX)
    target_link_libraries(ICEngine PUBLIC ws2_32 winmm user32 gdi32 shell32 dbghelp)
  else()
    find_package(X11 REQUIRED)
    find_package(Threads REQUIRED)
    target_compile_definitions(ICEngine PUBLIC GL_GLEXT_PROTOTYPES LINUX)
    target_include_directories(ICEngine PUBLIC ${X11_INCLUDE_DIR})
    target_link_libraries(ICEngine PUBLIC ${X11_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
endif()
