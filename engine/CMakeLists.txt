project(Engine)
include(../cmake/ICEData.cmake)


file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "/data/|/platform/|/input/input_(osx|win|x11)\\.cpp$|/io/file_win\\.cpp$|_(dx9|dx11|gles|gles2)\\.cpp$|/net/companion_(link|web)\\.cpp$|/input/companion(_room)?\\.cpp$|/util/join_qr\\.cpp$")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS 3rdparty/Lua/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
list(APPEND ENGINE_SOURCES src/platform/window.cpp src/platform/application.cpp src/platform/platform_events.cpp)
if(APPLE)
  list(APPEND ENGINE_SOURCES src/platform/macos/window.mm src/input/input_osx.cpp)
elseif(WIN32)
  list(APPEND ENGINE_SOURCES src/platform/win32/window.cpp src/input/input_win.cpp src/io/file_win.cpp)
else()
  list(APPEND ENGINE_SOURCES src/platform/x11/window.cpp src/input/input_x11.cpp)
endif()
# Asset ingestion and orbit math can run in headless tools without a graphics context.
list(REMOVE_ITEM ENGINE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/animation/scene_animation.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/io/resource_uri.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/assets/scene_asset.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/assets/gltf_readers.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/render/scene_camera.cpp")
add_library(ICEAssetRuntime STATIC src/animation/scene_animation.cpp src/assets/scene_asset.cpp src/assets/gltf_readers.cpp src/render/scene_camera.cpp)
target_link_libraries(ICEAssetRuntime PUBLIC ice::math ice::uri)
target_include_directories(ICEAssetRuntime PUBLIC include)
target_include_directories(ICEAssetRuntime SYSTEM PRIVATE 3rdparty)
target_compile_features(ICEAssetRuntime PUBLIC cxx_std_${ICE_CXX_STANDARD})
ice_strict_target(ICEAssetRuntime)
list(REMOVE_ITEM ENGINE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/meta/schema.cpp")
include(../cmake/ICEExpected.cmake)
add_library(ICEMetaRuntime STATIC src/meta/schema.cpp)
target_link_libraries(ICEMetaRuntime PUBLIC ice::expected)
add_library(ice::meta ALIAS ICEMetaRuntime)
target_include_directories(ICEMetaRuntime PUBLIC include)
target_compile_features(ICEMetaRuntime PUBLIC cxx_std_${ICE_CXX_STANDARD})
ice_strict_target(ICEMetaRuntime)
add_library(ICEngine ${ENGINE_SOURCES} ${LUA_SOURCES} 3rdparty/TLSF/tlsf.c 3rdparty/lea_malloc/leamalloc.c)
target_link_libraries(ICEngine PUBLIC ICEPlatformRuntime ICEAssetRuntime ICEMetaRuntime ICEDataRuntime)
ice_strict_target(ICEngine)
# This backend deliberately uses the macOS compatibility OpenGL/OpenAL APIs.
if(APPLE)
  target_compile_options(ICEngine PRIVATE -Wno-deprecated-declarations)
endif()
target_include_directories(ICEngine PUBLIC include PRIVATE src)
target_include_directories(ICEngine SYSTEM PUBLIC 3rdparty)
if(APPLE)
  target_compile_definitions(ICEngine PUBLIC GL_SILENCE_DEPRECATION)
  target_link_libraries(ICEngine PUBLIC "-framework AppKit" "-framework ApplicationServices" "-framework OpenGL" "-framework OpenAL" "-framework CoreText" "-framework CoreGraphics")
endif()

if(NOT APPLE)
  find_package(OpenGL REQUIRED)
  find_package(OpenAL REQUIRED)
  target_include_directories(ICEngine PUBLIC ${OPENAL_INCLUDE_DIR})
  target_link_libraries(ICEngine PUBLIC OpenGL::GL OpenGL::GLU ${OPENAL_LIBRARY})
  if(WIN32)
    target_compile_definitions(ICEngine PUBLIC ICE_OPENGL_ONLY WIN32 _CRT_SECURE_NO_WARNINGS NOMINMAX)
    target_link_libraries(ICEngine PUBLIC ws2_32 winmm user32 gdi32 shell32 dbghelp)
  else()
    find_package(X11 REQUIRED)
    find_package(Threads REQUIRED)
    target_compile_definitions(ICEngine PUBLIC GL_GLEXT_PROTOTYPES LINUX)
    target_include_directories(ICEngine PUBLIC ${X11_INCLUDE_DIR})
    target_link_libraries(ICEngine PUBLIC ${X11_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
endif()
