file(GLOB SKEEBALL_SOURCES CONFIGURE_DEPENDS Source/*.cpp)
list(FILTER SKEEBALL_SOURCES EXCLUDE REGEX "/wiimote\\.cpp$")
add_executable(SkeeBall MACOSX_BUNDLE ${SKEEBALL_SOURCES})
target_link_libraries(SkeeBall PRIVATE ICEngine SkeeBallPhysics)

    target_link_options(SkeeBall PRIVATE /ENTRY:mainCRTStartup)

endif()
