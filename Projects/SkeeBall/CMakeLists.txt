file(GLOB_RECURSE BULLET_SOURCES CONFIGURE_DEPENDS bullet/*.cpp)
add_library(SkeeBallPhysics STATIC ${BULLET_SOURCES})
target_include_directories(SkeeBallPhysics PUBLIC bullet)
file(GLOB SKEEBALL_SOURCES CONFIGURE_DEPENDS Source/*.cpp)
list(FILTER SKEEBALL_SOURCES EXCLUDE REGEX "/wiimote\\.cpp$")
add_executable(SkeeBall MACOSX_BUNDLE ${SKEEBALL_SOURCES})
target_link_libraries(SkeeBall PRIVATE ICEngine SkeeBallPhysics ICECompanionWeb)
target_include_directories(SkeeBall PRIVATE ../Common)
set_target_properties(SkeeBall PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" MACOSX_BUNDLE_GUI_IDENTIFIER "com.infinitecombustion.skeeball")
add_custom_target(SkeeBallAssets COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/GameFolder/Resource" "$<TARGET_FILE_DIR:SkeeBall>/Resource" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/GameFolder/settings.ini" "$<TARGET_FILE_DIR:SkeeBall>/settings.ini")
if(WIN32)
  set_target_properties(SkeeBall PROPERTIES WIN32_EXECUTABLE TRUE)
  if(MSVC)
    target_link_options(SkeeBall PRIVATE /ENTRY:mainCRTStartup)
  endif()
endif()

if(NOT APPLE)
    set_target_properties(SkeeBall PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/SkeeBall")
endif()

if(APPLE)
  target_sources(SkeeBall PRIVATE GameFolder/Resource/art/SkeeBall.icns)
  set_source_files_properties(GameFolder/Resource/art/SkeeBall.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(SkeeBall PROPERTIES MACOSX_BUNDLE_ICON_FILE SkeeBall.icns)
elseif(WIN32)
  enable_language(RC)
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/SkeeBall.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/SkeeBall.rc" @ONLY)
  target_sources(SkeeBall PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/SkeeBall.rc")
endif()

add_dependencies(SkeeBall SkeeBallAssets)
