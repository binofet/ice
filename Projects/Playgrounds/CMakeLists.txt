foreach(example OrbitalSalvage WarehouseStealth KartRacer VoxelSandbox)
    add_executable(${example} MACOSX_BUNDLE Source/Main.cpp)
    target_link_libraries(${example} PRIVATE ICEngine)
    target_include_directories(${example} PRIVATE ../Common)
    set_target_properties(${example} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" MACOSX_BUNDLE_GUI_IDENTIFIER "org.ice.${example}")
    if(WIN32)
        set_target_properties(${example} PROPERTIES WIN32_EXECUTABLE TRUE)
        if(MSVC)
            target_link_options(${example} PRIVATE /ENTRY:mainCRTStartup)
        endif()
    endif()
endforeach()
target_compile_definitions(OrbitalSalvage PRIVATE GAME_MODE=0)
target_compile_definitions(WarehouseStealth PRIVATE GAME_MODE=1)
target_compile_definitions(KartRacer PRIVATE GAME_MODE=2)
target_compile_definitions(VoxelSandbox PRIVATE GAME_MODE=3)
