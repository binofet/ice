cmake_minimum_required(VERSION 3.20)
project(ICEMobileProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT TARGET ICEPlatformRuntime)
    add_subdirectory(../../Platform platform)
endif()
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJCXX)
    add_executable(ICEMobileProbe MACOSX_BUNDLE ios/Main.mm)
    target_compile_options(ICEMobileProbe PRIVATE -fobjc-arc)
    target_link_libraries(ICEMobileProbe PRIVATE ICEPlatformRuntime "-framework UIKit" "-framework QuartzCore" "-framework Metal" "-framework CoreMotion")
    set_target_properties(ICEMobileProbe PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/ios/Info.plist" XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "org.ice.MobileProbe" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
elseif(ANDROID)
    add_library(ice_mobile_probe SHARED android/Main.cpp "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
    target_include_directories(ice_mobile_probe PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
    target_link_libraries(ice_mobile_probe PRIVATE ICEPlatformRuntime android log)
    target_link_options(ice_mobile_probe PRIVATE -uANativeActivity_onCreate)
else()
    message(FATAL_ERROR "Configure with an iOS or Android toolchain; desktop runtime tests are in the root project.")
endif()
