add_executable(EngineTest MACOSX_BUNDLE Source/Main.cpp Source/TestApp.cpp)
target_link_libraries(EngineTest PRIVATE ICEngine)
target_include_directories(EngineTest PRIVATE ../Common)
set_target_properties(EngineTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" MACOSX_BUNDLE_GUI_IDENTIFIER "com.infinitecombustion.enginetest")
add_custom_target(EngineTestAssets COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Resource" "$<TARGET_FILE_DIR:EngineTest>/Resource" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/engine/resources/shaders" "$<TARGET_FILE_DIR:EngineTest>/Resource/engine-shaders")
if(WIN32)
  set_target_properties(EngineTest PROPERTIES WIN32_EXECUTABLE TRUE)
  if(MSVC)
    target_link_options(EngineTest PRIVATE /ENTRY:mainCRTStartup)
  endif()
endif()

if(NOT APPLE)
    set_target_properties(EngineTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/EngineTest")
endif()

add_dependencies(EngineTest EngineTestAssets)
