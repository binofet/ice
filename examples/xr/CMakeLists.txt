cmake_minimum_required(VERSION 3.20)
project(ICEXRProbe LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "This optional OpenXR hardware probe currently targets Android headsets")
endif()
set(ICE_BUILD_COMPANION OFF CACHE BOOL "No companion networking required by the XR probe" FORCE)
include(../../cmake/ICEStandards.cmake)
include(../../cmake/ICEPlatform.cmake)
include(../../cmake/ICEMath.cmake)
include(../../cmake/ICEAudio.cmake)
set(ICE_OPENXR_AAR_DIR "" CACHE PATH "Extracted SHA256-verified Khronos OpenXR Android loader 1.1.63")
if(NOT EXISTS "${ICE_OPENXR_AAR_DIR}/jni/${ANDROID_ABI}/libopenxr_loader.so")
    message(FATAL_ERROR "Use examples/xr/android/build.sh to fetch and verify the pinned Khronos loader")
endif()
add_library(OpenXRLoader SHARED IMPORTED)
set_target_properties(OpenXRLoader PROPERTIES
    IMPORTED_LOCATION "${ICE_OPENXR_AAR_DIR}/jni/${ANDROID_ABI}/libopenxr_loader.so"
    INTERFACE_INCLUDE_DIRECTORIES "${ICE_OPENXR_AAR_DIR}/prefab/modules/headers/include")
add_library(ice_xr_probe SHARED android/main.cpp "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
target_include_directories(ice_xr_probe PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
target_link_libraries(ice_xr_probe PRIVATE ice::platform ice::math ice::audio OpenXRLoader android log EGL GLESv3)
target_compile_definitions(ice_xr_probe PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_link_options(ice_xr_probe PRIVATE -uANativeActivity_onCreate)
ice_strict_target(ice_xr_probe)
