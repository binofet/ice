cmake_minimum_required(VERSION 3.22)
project(ICEXRProbe LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "This optional OpenXR hardware probe currently targets Android headsets")
endif()
get_filename_component(ICE_XR_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(feature COMPANION PHYSICS_2D PHYSICS_3D SCRIPTING COMMANDS DESKTOP OPENGL METAL TOOLS)
    set(ICE_BUILD_${feature} OFF CACHE BOOL "Optional XR renderer/lifecycle lab" FORCE)
endforeach()
set(ICE_BUILD_GLES ON CACHE BOOL "Shared embedded rendering profile" FORCE)
set(ICE_BUILD_AUDIO ON CACHE BOOL "Explicit native push-to-talk" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "No root test executables in the APK" FORCE)
include("${ICE_XR_SOURCE_ROOT}/cmake/ICEStandards.cmake")
add_subdirectory("${ICE_XR_SOURCE_ROOT}" ice)
include(android/Host.cmake)
add_library(ice_xr_probe SHARED android/main.cpp)
target_link_libraries(ice_xr_probe PRIVATE ice::xr_android_host ice::assets)
ice_strict_target(ice_xr_probe)
