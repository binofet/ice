cmake_minimum_required(VERSION 3.20)
project(ICEMobileProbe LANGUAGES C CXX)
include(../../cmake/ICEStandards.cmake)
if(NOT TARGET ICEPlatformRuntime)
    include(../../cmake/ICEPlatform.cmake)
endif()
set(ICE_MOBILE_BUNDLE_ID "org.ice.MobileProbe" CACHE STRING "Bundle identifier for development signing")
set(ICE_DEVELOPMENT_TEAM "" CACHE STRING "Apple development team ID; empty builds unsigned")
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "watchOS")
    enable_language(OBJCXX Swift)
    add_executable(ICEWatchProbe MACOSX_BUNDLE watchos/Main.swift watchos/Bridge.mm)
    target_compile_options(ICEWatchProbe PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(ICEWatchProbe PRIVATE ice::companion "-framework SwiftUI" "-framework WatchKit" "-framework CoreMotion")
    set_target_properties(ICEWatchProbe PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/watchos/Info.plist"
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${ICE_MOBILE_BUNDLE_ID}"
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "4"
        XCODE_ATTRIBUTE_SWIFT_VERSION "5.0"
        XCODE_ATTRIBUTE_SWIFT_OBJC_BRIDGING_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/watchos/Bridge.h"
        XCODE_GENERATE_SCHEME YES)
    if(ICE_DEVELOPMENT_TEAM)
        set_target_properties(ICEWatchProbe PROPERTIES XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${ICE_DEVELOPMENT_TEAM}" XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Automatic")
    endif()
elseif(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJCXX)
    add_executable(ICEMobileProbe MACOSX_BUNDLE ios/Main.mm ios/Assets.xcassets)
    set_source_files_properties(ios/Assets.xcassets PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(ICEMobileProbe PROPERTIES XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME AppIcon)
    target_compile_options(ICEMobileProbe PRIVATE -fobjc-arc)
    target_link_libraries(ICEMobileProbe PRIVATE ice::companion "-framework UIKit" "-framework QuartzCore" "-framework Metal" "-framework CoreMotion" "-framework CoreGraphics")
    set_target_properties(ICEMobileProbe PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/ios/Info.plist" XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${ICE_MOBILE_BUNDLE_ID}" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_GENERATE_SCHEME YES)
    if(ICE_DEVELOPMENT_TEAM AND CMAKE_OSX_SYSROOT MATCHES "iphoneos|iPhoneOS")
        set_target_properties(ICEMobileProbe PROPERTIES XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${ICE_DEVELOPMENT_TEAM}" XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Automatic" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "YES")
    endif()
elseif(ANDROID)
    add_library(ice_mobile_probe SHARED android/Main.cpp "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
    target_include_directories(ice_mobile_probe PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
    target_link_libraries(ice_mobile_probe PRIVATE ice::companion android log)
    target_link_options(ice_mobile_probe PRIVATE -uANativeActivity_onCreate)
else()
    message(FATAL_ERROR "Configure with an iOS or Android toolchain; desktop runtime tests are in the root project.")
endif()
