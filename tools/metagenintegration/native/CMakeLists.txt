option(ICE_METAGEN_RTTR "Build the optional generated RTTR integration proof" OFF)
if(ICE_METAGEN_RTTR)
    include(FetchContent)
    # Optional development dependency, exact upstream source and archive digest.
    # FETCHCONTENT_SOURCE_DIR_RTTR may point at a previously verified offline copy.
    set(BUILD_RTTR_DYNAMIC OFF CACHE BOOL "" FORCE)
    set(BUILD_STATIC ON CACHE BOOL "" FORCE)
    foreach(flag BUILD_UNIT_TESTS BUILD_BENCHMARKS BUILD_EXAMPLES BUILD_DOCUMENTATION BUILD_INSTALLER BUILD_PACKAGE USE_PCH)
        set(${flag} OFF CACHE BOOL "" FORCE)
    endforeach()
    FetchContent_Declare(rttr
        URL https://codeload.github.com/rttrorg/rttr/tar.gz/7edbd580cfad509a3253c733e70144e36f02ecd4
        URL_HASH SHA256=bba4b6fac2349fa6badc701aad5e7afb87504a7089a867b1a7cbed08fb2f3a90
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_GetProperties(rttr)
    if(NOT rttr_POPULATED)
        FetchContent_Populate(rttr)
    endif()
    # Preserve the downloaded/offline upstream tree. Patch only our private build copy.
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(rttr_patched "${CMAKE_CURRENT_BINARY_DIR}/rttr-patched")
    file(COPY "${rttr_SOURCE_DIR}/" DESTINATION "${rttr_patched}")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/patch_rttr.py" "${rttr_patched}"
                    RESULT_VARIABLE patch_result)
    if(NOT patch_result EQUAL 0)
        message(FATAL_ERROR "Pinned RTTR compatibility patch failed")
    endif()
    add_subdirectory("${rttr_patched}" "${CMAKE_CURRENT_BINARY_DIR}/rttr-build" EXCLUDE_FROM_ALL)
    set_property(TARGET rttr_core_lib PROPERTY SYSTEM TRUE)
    add_executable(meta_rttr_tests rttr_tests.cpp rttr_adapter.cpp ../generated/rttr.generated.cpp)
    target_include_directories(meta_rttr_tests PRIVATE ../generated)
    target_link_libraries(meta_rttr_tests PRIVATE meta_runtime RTTR::Core_Lib)
    ice_strict_target(meta_rttr_tests)
    add_test(NAME meta_rttr COMMAND meta_rttr_tests)
endif()
