cmake_minimum_required(VERSION 3.22)
project(ICEActionRouter LANGUAGES CXX)
get_filename_component(ICE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ICE_BUILD_COMPANION OFF CACHE BOOL "Portable reducer/router only" FORCE)
include("${ICE_ROOT}/cmake/ICEStandards.cmake")
include("${ICE_ROOT}/cmake/ICEExpected.cmake")
include("${ICE_ROOT}/cmake/ICEPlatform.cmake")
include("${ICE_ROOT}/cmake/ICEInput.cmake")
add_library(ICEActionRouterRuntime STATIC "${ICE_ROOT}/engine/src/input/action_router.cpp")
add_library(ice::action_router ALIAS ICEActionRouterRuntime)
target_link_libraries(ICEActionRouterRuntime PUBLIC ice::input)
ice_strict_target(ICEActionRouterRuntime)
add_executable(ice_action_router_tests ../action_router.cpp allocations.cpp)
target_link_libraries(ice_action_router_tests PRIVATE ice::action_router)
ice_strict_target(ice_action_router_tests)
enable_testing()
foreach(case profile held delta cancel generation consumers atomic clocks allocation)
    add_test(NAME ice_action_router_${case} COMMAND ice_action_router_tests ${case})
    set_tests_properties(ice_action_router_${case} PROPERTIES TIMEOUT 30)
endforeach()
