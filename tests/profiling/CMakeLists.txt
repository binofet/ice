cmake_minimum_required(VERSION 3.22)
project(ICEProfilingTests LANGUAGES CXX)
get_filename_component(ICE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(ICE_PROFILE_RECORDING_TEST "Private in-memory test backend, never a production selection" OFF)
foreach(feature PHYSICS_2D PHYSICS_3D AUDIO AUDIO_PLAYBACK AUDIO_SPEECH ANDROID_SPEECH
        COMPANION DESKTOP OPENGL METAL GLES TOOLS COMMANDS)
    set(ICE_BUILD_${feature} OFF CACHE BOOL "CPU profiling fixture" FORCE)
endforeach()
set(ICE_BUILD_SCRIPTING ON CACHE BOOL "Actual Lua integration" FORCE)
if(ICE_PROFILE_RECORDING_TEST)
    set(ICE_ENABLE_PROFILING OFF CACHE BOOL "Recording harness replaces only test target dependencies" FORCE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Focused CPU tests only" FORCE)
add_subdirectory("${ICE_ROOT}" ice)
enable_testing()
if(ICE_PROFILE_RECORDING_TEST)
    add_library(ICEProfilingRecording STATIC "${ICE_ROOT}/engine/src/profiling/profile.cpp" recording_backend.cpp)
    target_include_directories(ICEProfilingRecording PUBLIC "${ICE_ROOT}/engine/include" PRIVATE "${ICE_ROOT}/engine/src/profiling")
    target_compile_definitions(ICEProfilingRecording PUBLIC ICE_PROFILING_ENABLED=1 ICE_PROFILING_RECORDING_TEST=1)
    target_compile_features(ICEProfilingRecording PUBLIC cxx_std_${ICE_CXX_STANDARD})
    ice_strict_target(ICEProfilingRecording)
    target_link_libraries(ICEScriptLua PRIVATE ICEProfilingRecording)
    set(profile_target ICEProfilingRecording)
else()
    set(profile_target ice::profiling)
endif()
add_executable(ice_profiling_tests ../profiling.cpp allocations.cpp)
target_link_libraries(ice_profiling_tests PRIVATE ice::script_lua ${profile_target})
ice_strict_target(ice_profiling_tests)
add_test(NAME ice_profiling_cpu COMMAND ice_profiling_tests)
foreach(test script_lua lua_environment)
    add_executable(ice_${test}_tests "${ICE_ROOT}/tests/${test}.cpp")
    target_link_libraries(ice_${test}_tests PRIVATE ice::script_lua)
    ice_strict_target(ice_${test}_tests)
    add_test(NAME ice_${test} COMMAND ice_${test}_tests)
endforeach()
set_tests_properties(ice_profiling_cpu ice_script_lua ice_lua_environment PROPERTIES TIMEOUT 120)
