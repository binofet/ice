cmake_minimum_required(VERSION 3.22)
project(ICEMacModifierProducer LANGUAGES CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "This CPU producer fixture requires the AppKit SDK")
endif()
get_filename_component(ICE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ICE_BUILD_COMPANION OFF CACHE BOOL "No services or native endpoints" FORCE)
include("${ICE_ROOT}/cmake/ICEStandards.cmake")
include("${ICE_ROOT}/cmake/ICEExpected.cmake")
include("${ICE_ROOT}/cmake/ICEPlatform.cmake")
include("${ICE_ROOT}/cmake/ICEDesktop.cmake")
include("${ICE_ROOT}/cmake/ICEInput.cmake")
add_executable(ice_platform_modifiers_macos_tests ../platform_modifiers_macos.mm)
target_link_libraries(ice_platform_modifiers_macos_tests PRIVATE ice::window ice::input "-framework AppKit")
target_compile_options(ice_platform_modifiers_macos_tests PRIVATE -fobjc-arc)
ice_strict_target(ice_platform_modifiers_macos_tests)
enable_testing()
add_test(NAME ice_platform_modifiers_macos COMMAND ice_platform_modifiers_macos_tests)
foreach(name platform_physical_input input_actions)
    add_executable(ice_${name}_tests "../${name}.cpp")
    target_link_libraries(ice_${name}_tests PRIVATE ice::window ice::input)
    ice_strict_target(ice_${name}_tests)
    add_test(NAME ice_${name} COMMAND ice_${name}_tests)
endforeach()
set_tests_properties(ice_platform_modifiers_macos ice_platform_physical_input ice_input_actions
    PROPERTIES TIMEOUT 30 LABELS "cpu;no_window")
