cmake_minimum_required(VERSION 3.22)
project(ice_static_pose_batch LANGUAGES CXX)
get_filename_component(ICE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(feature PHYSICS_2D SCRIPTING COMMANDS AUDIO COMPANION DESKTOP OPENGL GLES METAL TOOLS)
    set(ICE_BUILD_${feature} OFF CACHE BOOL "CPU-only static pose proof" FORCE)
endforeach()
set(ICE_BUILD_PHYSICS_3D ON CACHE BOOL "Native Jolt proof" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Only named focused fixtures" FORCE)
include("${ICE_ROOT}/cmake/ICEStandards.cmake")
add_subdirectory("${ICE_ROOT}" ice)
target_compile_definitions(ICEPhysics3D PRIVATE ICE_PHYSICS_POSE_BATCH_TESTING)
enable_testing()
foreach(fixture physics3d character3d static_collision3d)
    add_executable(ice_${fixture}_tests "${ICE_ROOT}/tests/${fixture}.cpp")
    target_link_libraries(ice_${fixture}_tests PRIVATE ice::physics3d)
    ice_strict_target(ice_${fixture}_tests)
    add_test(NAME ice_${fixture} COMMAND ice_${fixture}_tests)
    set_tests_properties(ice_${fixture} PROPERTIES TIMEOUT 30)
endforeach()

add_executable(ice_pose_batch_tests "${ICE_ROOT}/tests/physics3d_pose_batch.cpp")
target_compile_definitions(ice_pose_batch_tests PRIVATE ICE_PHYSICS_POSE_BATCH_TESTING)
target_link_libraries(ice_pose_batch_tests PRIVATE ice::physics3d Jolt)
get_target_property(JOLT_INCLUDES Jolt INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(ice_pose_batch_tests SYSTEM PRIVATE ${JOLT_INCLUDES})
ice_strict_target(ice_pose_batch_tests)
# Compile the identical production module with all fixture controls absent as well.
add_library(ice_pose_production_compile OBJECT "${ICE_ROOT}/engine/src/physics/world3d.cpp")
target_link_libraries(ice_pose_production_compile PRIVATE ice::physics_core Jolt)
target_include_directories(ice_pose_production_compile SYSTEM PRIVATE ${JOLT_INCLUDES})
ice_strict_target(ice_pose_production_compile)
foreach(case success validation rollback lifetimes epochs wake characters prepare_failures)
    add_test(NAME ice_pose_batch_${case} COMMAND ice_pose_batch_tests ${case})
    set_tests_properties(ice_pose_batch_${case} PROPERTIES TIMEOUT 30)
endforeach()
