cmake_minimum_required(VERSION 3.22)
project(ICEMotionActionConsumer LANGUAGES CXX)
get_filename_component(ICE_DEFAULT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ICE_SOURCE_DIR "${ICE_DEFAULT_SOURCE}" CACHE PATH "Ordinary ICE add_subdirectory source")
include("${ICE_SOURCE_DIR}/cmake/ICEStandards.cmake")
# Exercise ordinary portable production registration without optional backend downloads.
set(BUILD_TESTING OFF CACHE BOOL "External consumer owns its tests" FORCE)
foreach(feature DESKTOP METAL OPENGL GLES COMPANION TOOLS PHYSICS_2D PHYSICS_3D SCRIPTING AUDIO COMMANDS)
    set(ICE_BUILD_${feature} OFF CACHE BOOL "Portable consumer fixture" FORCE)
endforeach()
add_subdirectory("${ICE_SOURCE_DIR}" ice)
foreach(consumer modules aggregate)
    add_executable(ice_consumer_${consumer} ../motion_action_consumer.cpp)
    if(consumer STREQUAL "modules")
        target_link_libraries(ice_consumer_${consumer} PRIVATE ice::action_router ice::motion)
    else()
        target_link_libraries(ice_consumer_${consumer} PRIVATE ice::engine)
    endif()
    ice_strict_target(ice_consumer_${consumer})
endforeach()
enable_testing()
foreach(consumer modules aggregate)
    add_test(NAME ice_consumer_${consumer} COMMAND ice_consumer_${consumer})
    set_tests_properties(ice_consumer_${consumer} PROPERTIES TIMEOUT 30 LABELS "cpu")
endforeach()
